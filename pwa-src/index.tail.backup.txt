unction enviarWARelatorio(){
  const bruto=S.din+S.pix+S.fiad;
  const liqTotal=bruto-S.desp;
  const totalSacos=S.esc+S.filt;
  const clisAtendidos=CLIENTES.filter(n=>S.atendidos.has(n));

  let linhasClientes='';
  clisAtendidos.forEach((nome,idx)=>{
    const vs=S.vpc[nome]||[];
    const itens=vs.map(v=>{
      const t=v.tipo==='esc'?'Escamas':v.tipo==='filt'?'Filtrado':'Observação';
      const p=v.pag==='PIX'?'PIX':v.pag==='Fiado'?'Fiado':'Pago/Dinheiro';
      return `   • ${v.qtd}x ${t} | ${p} | ${fmt(v.valor)}`;
    }).join('\n');
    const total=vs.reduce((a,v)=>a+v.valor,0);
    linhasClientes+=`${idx+1}. *${nome}*\n${itens}\n   Total: *${fmt(total)}*\n\n`;
  });

  const despesas=S.despDia.length
    ? S.despDia.map(d=>`• ${d}`).join('\n')
    : '• Nenhuma despesa registrada';

  const msg=`*GELO TUTOIA | FECHAMENTO DO DIA*
📅 ${dataSimples()}

📦 *QUANTIDADE VENDIDA*
❄️ Escamas: *${S.esc} sacos*
💎 Filtrado: *${S.filt} sacos*
📦 Total: *${totalSacos} sacos*

👥 *VENDAS POR CLIENTE*
${linhasClientes || 'Nenhuma venda registrada.\n'}
💰 *VALORES DO DIA*
💵 Dinheiro / Pago: *${fmt(S.din)}*
📱 PIX: *${fmt(S.pix)}*
💳 Fiado: *${fmt(S.fiad)}*
💰 Total de vendas: *${fmt(bruto)}*

📉 *DESPESAS*
${despesas}
Total despesas: *${fmt(S.desp)}*

✅ *FECHAMENTO*
Total após despesas: *${fmt(liqTotal)}*

_Gelo Tutoia_`;

  const a=document.getElementById('wa-link');
  a.href='https://wa.me/5521981378219?text='+encodeURIComponent(msg);
  a.click();
}
